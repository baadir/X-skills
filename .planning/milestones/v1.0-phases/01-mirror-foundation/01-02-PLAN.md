twikit Client initialization and auth (from RESEARCH.md Pattern 1):
```python
from twikit import Client, TooManyRequests
import asyncio

client = Client('en-US')

async def get_authenticated_client(cookies_file: str, username: str, email: str, password: str) -> Client:
    """Load cookies if available; fall back to login with enable_ui_metrics=True."""
    if Path(cookies_file).exists():
        try:
            client.load_cookies(cookies_file)
            await client.user_id()   # Lightweight validation call
            return client
        except Exception:
            pass  # Stale cookies — fall through to login
    await client.login(
        auth_info_1=username,
        auth_info_2=email,
        password=password,
        cookies_file=cookies_file,
        enable_ui_metrics=True,
    )
    return client
```

Incremental fetch with client-side cutoff (from RESEARCH.md Pattern 2):
```python
async def fetch_tweets_since(client: Client, user_id: str, cutoff: datetime) -> list[dict]:
    tweets = await client.get_user_tweets(user_id, 'Tweets', count=40)
    collected = []
    while tweets:
        stop = False
        for tweet in tweets:
            if tweet.created_at_datetime < cutoff:
                stop = True
                break
            collected.append(tweet)
        if stop:
            break
        try:
            tweets = await safe_next(tweets)
        except Exception:
            break
    return collected
```

Rate limit handler (from RESEARCH.md Pattern 4):
```python
from twikit import TooManyRequests
import time

async def safe_next(result):
    for attempt in range(5):
        try:
            return await result.next()
        except TooManyRequests as e:
            reset_at = getattr(e, 'rate_limit_reset', None)
            wait = max(0, reset_at - int(time.time())) + 5 if reset_at else 60 * (2 ** attempt)
            await asyncio.sleep(wait)
    raise RuntimeError("Max rate limit retries exceeded")
```
