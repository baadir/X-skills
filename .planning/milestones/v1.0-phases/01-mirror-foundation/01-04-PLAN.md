async def run_sync(dry_run: bool = False, since: str | None = None) -> None:
    # 1. Load config
    config = load_env()

    # 2. Load style_memory (with seed applied on first run)
    style_memory = load_style_memory()

    # 3. Determine cutoff
    if since:
        cutoff = datetime.fromisoformat(since).replace(tzinfo=timezone.utc)
    elif style_memory.get("last_synced_at"):
        cutoff = datetime.fromisoformat(style_memory["last_synced_at"]).replace(tzinfo=timezone.utc)
    else:
        cutoff = None  # First run -- fetch_tweets_since handles None cutoff

    # 4. Authenticate
    client = await get_authenticated_client(config)

    # 5. Fetch
    raw_tweets = await fetch_tweets_since(client, config["USER_ID"], cutoff)

    # 6-9. Categorize, archive, high_performers, snapshot (same logic as before)
    ...

    # 10. evolution_log
    if not dry_run and new_tweets:
        hp_delta = len(new_high_performers) - prev_hp_count
        append_evolution_log(
            style_memory,
            trigger="sync" if not since else "--since override",
            summary=f"Added {len(new_tweets)} tweets. High_performers delta: {hp_delta:+d}.",
        )

    # 11. Save (atomic -- skip in dry-run)
    if not dry_run:
        save_style_memory(style_memory)

    # 12. Report
    _print_sync_report(...)


def main() -> None:
    args = parse_args()
    if args.verbose:
        logging.getLogger().setLevel(logging.DEBUG)
    try:
        asyncio.run(run_sync(dry_run=args.dry_run, since=args.since))
    except KeyboardInterrupt:
        logger.info("Kullanici tarafindan iptal edildi.")
        sys.exit(0)
    except Exception as e:
        logger.exception("Beklenmeyen hata: %s", e)
        sys.exit(1)
